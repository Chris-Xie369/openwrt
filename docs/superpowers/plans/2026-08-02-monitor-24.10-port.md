```c
#include <pthread.h>
#include <stddef.h>

/* monitor 唯一调用的 pwrapper 函数：变参线程创建。
 * 3 处调用（main.cpp:141/163/179）全传 NULL 参数（无线程 arg），
 * 简化为 pthread_create(funcptr, NULL)。priority 忽略（无实时调度需求）。 */
int pthreadSpawn(pthread_t *ptid, int priority, size_t stacksize,
                 void *funcptr, unsigned args, ...)
{
	pthread_attr_t attr;
	pthread_attr_init(&attr);
	if (stacksize)
		pthread_attr_setstacksize(&attr, stacksize);
	int ret = pthread_create(ptid, &attr, (void *(*)(void *))funcptr, NULL);
	pthread_attr_destroy(&attr);
	return ret;
}
```

```makefile
  SECTION:=utils

  TITLE:=Gateway monitor daemon (MQTT heartbeat + watchdog)
  DEPENDS:=+libpaho-mqtt-c +libopenssl +zlib +libcurl +libpthread

define Package/monitor/description
 C monitor daemon for HLK-7688A gateway: MQTT heartbeat uplink, flock
 singleton, crontab watchdog every 2 min. Cross-compiled from external

# monitor internal include paths (mirror of its Makefile HEADER_PATH ./inc/... part)
MONITOR_INC := \
	-I./inc/base -I./inc/util -I./inc/third_party/zlib/ \
	-I./inc/third_party/cjson/ -I./inc/third_party/posix/ \
	-I./inc/http -I./inc/iota/base -I./inc/iota/agentlite \
	-I./inc/iota/service -I./inc/iota/protocol -I./inc/iota/conn \
	-I./inc/periph -I./inc/common -I./inc/file \

MONITOR_LIBS := \
	-L$(STAGING_DIR)/usr/lib \
	-lpaho-mqtt3as -lssl -lcrypto -lz -lrt -lm -lcurl -lpthread

define Build/Compile
	# Remove stale artifacts from any prior (old toolchain) build
	$(MAKE) -C $(PKG_BUILD_DIR) clean || true

	# Build with full toolchain override (GNU make cmdline > Makefile '=' assignment)
	$(MAKE) -C $(PKG_BUILD_DIR) all \
		CC="$(TARGET_CC)" \
		CXX="$(TARGET_CXX)" \
		CFLAGS="$(TARGET_CFLAGS) -std=c99 -fPIC -fsigned-char -ffunction-sections -fdata-sections -w" \
		CXXFLAGS="$(TARGET_CXXFLAGS) -std=c++11 -fPIC -fexceptions -fsigned-char -ffunction-sections -fdata-sections -w" \

		LIB_PATH="-L$(STAGING_DIR)/usr/lib" \
		LIBS="$(MONITOR_LIBS)"

	$(INSTALL_BIN) $(PKG_BUILD_DIR)/monitor $(1)/usr/bin/monitor
endef

$(eval $(call BuildPackage,monitor))
```

### Task 4: 编译验证 + 调试循环（核心）

> monitor 从未在 24.10 编译过，预期可能遇到 uClibc→musl 兼容性错误。此 task 是迭代调试循环。

**Files:**
- 无新文件；验证 Task 1-3 的编译结果

- [ ] **Step 1: 首次编译尝试**

```bash
make package/monitor/compile V=s 2>&1 | tee /tmp/monitor-build.log | tail -50
```

- [ ] **Step 2: 分析编译结果**

三种可能：
- **A) 编译成功**：monitor 二进制生成，跳到 Step 5
- **B) 编译错误**（.o 编译阶段）：uClibc→musl API 差异（头文件/函数签名）→ Step 3
- **C) 链接错误**（undefined reference）：缺失符号/库 → Step 4

- [ ] **Step 3: 编译错误修复循环（如 B）**

常见 musl 差异及修法：
- `error: unknown type name 'uint8_t'` 等 → 缺 `#include <stdint.h>`（uClibc 隐式包含，musl 严格）。修法：monitor 源码加 include（违背零改原则但必要），或 CFLAGS 加 `-include stdint.h`
- `implicit declaration of function 'xxx'` → 缺头文件。同上
- `error: 'PTHREAD_MUTEX_ERRORCHECK' undeclared` → musl 用 `PTHREAD_MUTEX_ERRORCHECK NP`。CFLAGS 补 define 或改源码
- 每个错误修复后重跑 `make package/monitor/compile V=s`

> 原则：优先用 CFLAGS 宏（`-Dxxx`）解决，避免改 .c/.cpp 源码。若必须改源码，最小改动 + 注释说明。

- [ ] **Step 4: 链接错误修复循环（如 C）**

常见缺失符号及修法：
- `undefined reference to 'pthreadMutexLock'` 等 pwrapper 函数 → pwrapper.c 补该函数实现
- `undefined reference to 'uci_xxx'` → MONITOR_LIBS 加 `-luci -lubox`（24.10 staging_dir 有）
- `undefined reference to 'mq_open'` → MONITOR_LIBS 加 `-lrt`（musl mqueue 在 rt）
- 每次修 Makefile 后重跑 compile
